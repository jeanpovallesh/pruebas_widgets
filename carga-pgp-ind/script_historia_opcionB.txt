// ============================================================================
// PROTOTIPO OPCIÓN B — carga directa desde la historia (SAC Analytics Designer)
// Widget: CargaPgp_1 (v1.0.3, sin JSON)  Tabla de planificación oculta: Table_Carga
//                                        Botón: Button_Cargar   Texto de estado: Text_Estado
// Data Action de limpieza: DA_LimpiarVersionTiempo (parámetros: P_VERSION, P_TIEMPO)
// El widget entrega TEXTO plano separado por "|"; no se usa JSON en ningún punto.
// Es un PROTOTIPO: probar primero con valido.csv (40 filas) antes de usar archivos grandes.
// ============================================================================

// --- 1) Evento onFileValidated del widget: habilita el botón solo si el archivo es válido
// CargaPgp_1.onFileValidated():
var okArchivo = CargaPgp_1.isValid() && !CargaPgp_1.hasPipeInValues();
Button_Cargar.setEnabled(okArchivo);
if (okArchivo) {
  Text_Estado.applyText(CargaPgp_1.getRowCount() + " filas listas. Pulsa 'Cargar al modelo'.");
} else if (CargaPgp_1.isValid()) {
  Text_Estado.applyText("Algún valor contiene el carácter |; corrígelo en el archivo.");
} else {
  Text_Estado.applyText("Archivo con " + CargaPgp_1.getErrorCount() + " error(es). No se puede cargar.");
}

// --- 2) Evento onClick del botón "Cargar al modelo"
// Button_Cargar.onClick():
if (!CargaPgp_1.isValid()) { return; }

var TAM = 500;                                   // filas por submit
var total = CargaPgp_1.getRowCount();
var versiones = CargaPgp_1.getVersionsText().split("|");
var periodos = CargaPgp_1.getPeriodsText().split("|");

// 2a) Reemplazo: borrar lo existente en Version x TIEMPO del archivo, para no duplicar si se carga dos veces.
//     (setUserInput sobrescribe celdas, pero no borra las que ya no vienen en el archivo.)
for (var v = 0; v < versiones.length; v++) {
  for (var p = 0; p < periodos.length; p++) {
    DA_LimpiarVersionTiempo.setParameterValue("P_VERSION", versiones[v]);
    DA_LimpiarVersionTiempo.setParameterValue("P_TIEMPO", periodos[p]);
    DA_LimpiarVersionTiempo.execute();
  }
}

// 2b) Escribir fila por fila en la tabla de planificación, con un submit cada TAM filas
var escritas = 0;
var fallidas = 0;
for (var i = 0; i < total; i++) {
  // f = [Version, TIEMPO, CECO_PGP, AUDITORIA_PGP, EMPLEADOS_PGP_IND, CUENTA_PGP_IND, ENTIDAD_PGP, MONTO]
  var f = CargaPgp_1.getRowText(i).split("|");
  var sel = {
    "Version": f[0],
    "TIEMPO": f[1],
    "CECO_PGP": f[2],
    "AUDITORIA_PGP": f[3],
    "EMPLEADOS_PGP_IND": f[4],
    "CUENTA_PGP_IND": f[5],
    "ENTIDAD_PGP": f[6]
  };
  // Si el modelo tiene una sola medida, no hace falta "@MeasureDimension".
  // Si SAC lo pide, agregar:  sel["@MeasureDimension"] = "MONTO";
  var ok = Table_Carga.getPlanning().setUserInput(sel, f[7]);
  if (ok) { escritas = escritas + 1; } else { fallidas = fallidas + 1; }

  if ((i + 1) % TAM === 0 || i === total - 1) {
    Table_Carga.getPlanning().submitData();
    Text_Estado.applyText("Avance " + (i + 1) + " de " + total + " — escritas " + escritas + ", fallidas " + fallidas);
  }
}

// 2c) Publicar la versión (los cambios quedan en versión privada hasta publicar)
for (var v2 = 0; v2 < versiones.length; v2++) {
  var nombre = versiones[v2].replace("public.", "");
  Table_Carga.getPlanning().getPublicVersion(nombre).publish();
}

Text_Estado.applyText("Terminado: " + escritas + " filas escritas, " + fallidas + " rechazadas por la tabla.");
